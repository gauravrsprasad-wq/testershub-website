What a test looks like

An example in Java with Appium and TestNG:

Example: LoginTest.java (Appium)

@Test
public void userCanLogInWithOtp() {
    HomeScreen home = new LoginScreen(driver)
            .enterPhoneNumber("9876543210")
            .requestOtp()
            .enterOtp(TestOtp.valid())
            .submit();

    assertEquals(home.greeting(), "Welcome back, Asha");
}

@Test
public void loginWorksAfterTheNetworkDrops() {
    LoginScreen login = new LoginScreen(driver).enterPhoneNumber("9876543210");
    Network.goOffline(driver);
    assertEquals(login.requestOtp().errorMessage(), "No internet connection. Try again.");
    Network.goOnline(driver);
    assertTrue(login.requestOtp().isOtpScreenShown());
}
